-- =====================================================================
-- jurong_marine: add parent-child relationships (foreign keys)
-- Run each STEP in order. Do NOT run Step 3 until Steps 1-2 are clean.
-- Take a backup first:  mysqldump jurong_marine > jurong_marine_backup.sql
-- =====================================================================

USE jurong_marine;

-- ---------------------------------------------------------------------
-- STEP 0: Pre-flight checks
-- ---------------------------------------------------------------------

-- 0a. All tables must be InnoDB (MyISAM ignores foreign keys).
SELECT table_name, engine
FROM information_schema.tables
WHERE table_schema = 'jurong_marine';

-- 0b. Joined varchar columns must share the same charset/collation,
--     otherwise ADD CONSTRAINT fails with error 3780.
SELECT table_name, column_name, character_set_name, collation_name
FROM information_schema.columns
WHERE table_schema = 'jurong_marine'
  AND column_name IN ('machine_id', 'emp_code');


-- ---------------------------------------------------------------------
-- STEP 1: Find bad data that would block the constraints
--         Every query below should return 0 rows before Step 3.
-- ---------------------------------------------------------------------

-- 1a. employees.emp_code must be unique and not null (it becomes a parent key)
SELECT emp_code, COUNT(*) AS cnt
FROM employees
GROUP BY emp_code
HAVING emp_code IS NULL OR COUNT(*) > 1;

-- 1b. Sensor machine_ids longer than 10 chars (would be truncated in Step 2b)
SELECT DISTINCT machine_id, CHAR_LENGTH(machine_id) AS len
FROM vibration_temp_sample
WHERE CHAR_LENGTH(machine_id) > 10;

-- 1c. Orphan machine_ids (child rows pointing to non-existent machines)
SELECT 'work_orders' AS src, c.machine_id, COUNT(*) AS rows_affected
FROM work_orders c LEFT JOIN machines m ON m.machine_id = c.machine_id
WHERE c.machine_id IS NOT NULL AND m.machine_id IS NULL
GROUP BY c.machine_id
UNION ALL
SELECT 'safety_incidents', c.machine_id, COUNT(*)
FROM safety_incidents c LEFT JOIN machines m ON m.machine_id = c.machine_id
WHERE c.machine_id IS NOT NULL AND m.machine_id IS NULL
GROUP BY c.machine_id
UNION ALL
SELECT 'vibration_temp_sample', c.machine_id, COUNT(*)
FROM vibration_temp_sample c LEFT JOIN machines m ON m.machine_id = c.machine_id
WHERE m.machine_id IS NULL
GROUP BY c.machine_id;

-- 1d. Orphan emp_codes
SELECT 'work_orders' AS src, c.emp_code, COUNT(*) AS rows_affected
FROM work_orders c LEFT JOIN employees e ON e.emp_code = c.emp_code
WHERE c.emp_code IS NOT NULL AND e.emp_code IS NULL
GROUP BY c.emp_code
UNION ALL
SELECT 'safety_incidents', c.emp_code, COUNT(*)
FROM safety_incidents c LEFT JOIN employees e ON e.emp_code = c.emp_code
WHERE c.emp_code IS NOT NULL AND e.emp_code IS NULL
GROUP BY c.emp_code;

-- If 1c/1d return rows, either insert the missing parent rows into
-- machines/employees, correct the typo'd codes, or set them to NULL, e.g.:
--   UPDATE work_orders SET emp_code = 'E0123' WHERE emp_code = 'E123';


-- ---------------------------------------------------------------------
-- STEP 2: Fix parent keys and column types
-- ---------------------------------------------------------------------

-- 2a. Make emp_code a proper candidate key (required to be referenced)
ALTER TABLE employees
  MODIFY emp_code VARCHAR(8) NOT NULL,
  ADD CONSTRAINT uq_employees_emp_code UNIQUE (emp_code);

-- 2b. Align sensor machine_id with machines.machine_id (was VARCHAR(50))
--     Note: rebuilds the table; may take a while if it is large.
ALTER TABLE vibration_temp_sample
  MODIFY machine_id VARCHAR(10) NOT NULL;


-- ---------------------------------------------------------------------
-- STEP 3: Add the foreign keys
--   ON DELETE RESTRICT  -> can't delete a machine/employee that has history
--   ON UPDATE CASCADE   -> renaming a machine_id/emp_code updates children
--   MySQL auto-creates an index on each child column.
-- ---------------------------------------------------------------------

-- machines -> work_orders
ALTER TABLE work_orders
  ADD CONSTRAINT fk_wo_machine
    FOREIGN KEY (machine_id) REFERENCES machines (machine_id)
    ON DELETE RESTRICT ON UPDATE CASCADE,
  ADD CONSTRAINT fk_wo_employee
    FOREIGN KEY (emp_code) REFERENCES employees (emp_code)
    ON DELETE RESTRICT ON UPDATE CASCADE;

-- machines / employees -> safety_incidents
ALTER TABLE safety_incidents
  ADD CONSTRAINT fk_incident_machine
    FOREIGN KEY (machine_id) REFERENCES machines (machine_id)
    ON DELETE RESTRICT ON UPDATE CASCADE,
  ADD CONSTRAINT fk_incident_employee
    FOREIGN KEY (emp_code) REFERENCES employees (emp_code)
    ON DELETE RESTRICT ON UPDATE CASCADE;

-- machines -> vibration_temp_sample
ALTER TABLE vibration_temp_sample
  ADD CONSTRAINT fk_vib_machine
    FOREIGN KEY (machine_id) REFERENCES machines (machine_id)
    ON DELETE RESTRICT ON UPDATE CASCADE;


-- ---------------------------------------------------------------------
-- STEP 4: Verify
-- ---------------------------------------------------------------------
SELECT table_name, constraint_name, column_name,
       referenced_table_name, referenced_column_name
FROM information_schema.key_column_usage
WHERE table_schema = 'jurong_marine'
  AND referenced_table_name IS NOT NULL
ORDER BY table_name;
